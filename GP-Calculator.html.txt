/* =========================================================
   GET ELEMENTS
   ========================================================= */

var gst =
    document.getElementById("gst");

var tp =
    document.getElementById("tp");

var bonus =
    document.getElementById("bonus");

var mrp =
    document.getElementById("mrp");

var pts =
    document.getElementById("pts");

var calculate =
    document.getElementById("calculate");

var reset =
    document.getElementById("reset");

var error =
    document.getElementById("error");

var rPts =
    document.getElementById("rPts");

var rEffectiveTP =
    document.getElementById("rEffectiveTP");

var rGP =
    document.getElementById("rGP");

var rGPPercent =
    document.getElementById("rGPPercent");


/* =========================================================
   NUMBER FUNCTION
   ========================================================= */

function getNumber(element) {

    var value =
        parseFloat(element.value);

    if (isNaN(value)) {
        return null;
    }

    return value;

}


/* =========================================================
   CURRENCY FORMAT
   ========================================================= */

function money(value) {

    if (isNaN(value)) {
        value = 0;
    }

    return "₹ " +
        value.toLocaleString(
            "en-IN",
            {
                minimumFractionDigits: 2,
                maximumFractionDigits: 2
            }
        );

}


/* =========================================================
   CALCULATE PTS
   ========================================================= */

function calculatePTS() {

    var gstValue =
        getNumber(gst);

    var mrpValue =
        getNumber(mrp);


    if (
        gstValue === null ||
        mrpValue === null ||
        mrpValue <= 0
    ) {

        pts.value =
            "0.00";

        return 0;

    }


    /*
       PTS Formula:

       (MRP / (1 + GST%))
       × 80%
       × 90%
    */


    var ptsValue =
        (
            mrpValue /
            (1 + gstValue / 100)
        )
        *
        0.80
        *
        0.90;


    pts.value =
        ptsValue.toFixed(2);


    return ptsValue;

}


/* =========================================================
   CALCULATE GP
   ========================================================= */

function calculateGP() {

    error.style.display =
        "none";

    error.innerHTML =
        "";


    var gstValue =
        getNumber(gst);

    var tpValue =
        getNumber(tp);

    var bonusValue =
        getNumber(bonus);

    var mrpValue =
        getNumber(mrp);


    /* =====================================================
       VALIDATION
       ===================================================== */

    if (gstValue === null) {

        showError(
            "Please choose GST rate."
        );

        return;

    }


    if (
        tpValue === null ||
        tpValue < 0
    ) {

        showError(
            "Please enter a valid TP."
        );

        return;

    }


    if (
        bonusValue === null ||
        bonusValue < 0
    ) {

        showError(
            "Please enter Free Quantity %."
        );

        return;

    }


    if (
        mrpValue === null ||
        mrpValue <= 0
    ) {

        showError(
            "Please enter a valid MRP."
        );

        return;

    }


    /* =====================================================
       PTS
       ===================================================== */

    var ptsValue =
        calculatePTS();


    if (
        ptsValue <= 0
    ) {

        showError(
            "Unable to calculate PTS."
        );

        return;

    }


    /* =====================================================
       EFFECTIVE TP
       =====================================================

       Example:

       TP = ₹100
       Free Quantity = 10%

       Effective TP
       = 100 × 1.10
       = ₹110

       ===================================================== */

    var effectiveTP =
        tpValue *
        (
            1 +
            bonusValue / 100
        );


    /* =====================================================
       GP RUPEES
       ===================================================== */

    var gp =
        ptsValue -
        effectiveTP;


    /* =====================================================
       GP PERCENT
       ===================================================== */

    var gpPercent =
        (
            gp /
            ptsValue
        )
        *
        100;


    /* =====================================================
       DISPLAY
       ===================================================== */

    rPts.innerHTML =
        money(ptsValue);


    rEffectiveTP.innerHTML =
        money(effectiveTP);


    rGP.innerHTML =
        money(gp);


    rGPPercent.innerHTML =
        gpPercent.toFixed(2) +
        "%";

}


/* =========================================================
   ERROR
   ========================================================= */

function showError(message) {

    error.innerHTML =
        message;

    error.style.display =
        "block";

}


/* =========================================================
   AUTOMATIC PTS
   ========================================================= */

gst.addEventListener(
    "change",
    function () {

        calculatePTS();

        calculateIfPossible();

    }
);


mrp.addEventListener(
    "input",
    function () {

        calculatePTS();

        calculateIfPossible();

    }
);


/* =========================================================
   AUTO GP
   ========================================================= */

tp.addEventListener(
    "input",
    function () {

        calculateIfPossible();

    }
);


bonus.addEventListener(
    "input",
    function () {

        calculateIfPossible();

    }
);


/* =========================================================
   CALCULATE IF ALL DATA AVAILABLE
   ========================================================= */

function calculateIfPossible() {

    var gstValue =
        getNumber(gst);

    var tpValue =
        getNumber(tp);

    var bonusValue =
        getNumber(bonus);

    var mrpValue =
        getNumber(mrp);


    if (
        gstValue !== null &&
        tpValue !== null &&
        bonusValue !== null &&
        mrpValue !== null &&
        mrpValue > 0
    ) {

        calculateGP();

    }

}

   BUTTON
